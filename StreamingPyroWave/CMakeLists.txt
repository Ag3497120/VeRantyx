cmake_minimum_required(VERSION 3.20)
project(VeRantyxPyroWave LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
option(PYROWAVE_BUILD_METAL_CODEC "Build the standalone Metal codec" OFF)
foreach(name packet_batch wavelet)
  add_executable(test_${name} tests/test_${name}.cpp)
  target_compile_options(test_${name} PRIVATE -UNDEBUG -fsanitize=undefined -fno-sanitize-recover=undefined)
  target_link_options(test_${name} PRIVATE -fsanitize=undefined -fno-sanitize-recover=undefined)
endforeach()
add_test(NAME packet_batch COMMAND test_packet_batch)
foreach(mode default 53 97 invalid)
  if(mode STREQUAL "53")
    set(expect 1)
  else()
    set(expect 0)
  endif()
  add_test(NAME wavelet_${mode} COMMAND test_wavelet ${expect})
  if(mode STREQUAL "default")
    set_tests_properties(wavelet_${mode} PROPERTIES ENVIRONMENT "PYROWAVE_WAVELET=")
  else()
    set_tests_properties(wavelet_${mode} PROPERTIES ENVIRONMENT "PYROWAVE_WAVELET=${mode}")
  endif()
endforeach()
if(PYROWAVE_BUILD_METAL_CODEC)
  if(NOT APPLE)
    message(FATAL_ERROR "The Metal codec requires macOS")
  endif()
  add_subdirectory(metal)
endif()
