find_package(Threads REQUIRED)
add_library(streaming_sei STATIC receiver/sei.c)
function(streaming_test name source)
  add_executable(${name} ${source})
  target_link_libraries(${name} PRIVATE Threads::Threads)
  target_compile_options(${name} PRIVATE -UNDEBUG)
  if(STREAMING_UBSAN AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(${name} PRIVATE -fsanitize=undefined -fno-sanitize-recover=undefined)
    target_link_options(${name} PRIVATE -fsanitize=undefined -fno-sanitize-recover=undefined)
  endif()
  add_test(NAME ${name} COMMAND ${name})
  set_tests_properties(${name} PROPERTIES TIMEOUT 60)
endfunction()
